async function checkUserStatus(userData) {
  if (!userData) {
    // user tidak ditemukan → arahkan ke halaman register
    redirectTo('/register');
    return;
  }

  if (userData.account_status === 'pending_assessment') {
    // user terdaftar tapi belum isi assessment
    // jangan tampilkan dashboard sama sekali
    showPendingAssessmentPage(userData);
    return;

  // normal → tampilkan dashboard
  showDashboard(userData);
}

const { error } = await supabase

    // identitas

    saving_goals,             saving_goals_raw,
    emergency_fund_duration,  emergency_fund_raw,
    investment_status,        investment_status_raw,
    investment_instruments,   investment_instruments_raw,

    ewallets_used,            ewallets_used_raw,
    paylater_active,          paylater_active_raw,
    payment_method_ranking,
    limit_nominal,
  })
  .eq('primary_phone', currentUserPhone);

if (error) {
  showToast('Gagal menyimpan. Coba lagi.', 'error');
  return;
}

// Setelah berhasil:
// 1. Generate affiliate_code untuk user ini (sama seperti primary)
//    format: 4 huruf dari nama + 4 digit terakhir nomor HP
const name4  = answers.name.replace(/[^a-zA-Z]/g,'').toUpperCase().slice(0,4).padEnd(4,'X');
const phone4 = currentUserPhone.replace(/\D/g,'').slice(-4);
const affiliate_code = name4 + phone4;

await supabase
  .from('users')
  .update({ affiliate_code })
  .eq('primary_phone', currentUserPhone);

// 2. Reload userData dari Supabase (biar dapat data terbaru)
